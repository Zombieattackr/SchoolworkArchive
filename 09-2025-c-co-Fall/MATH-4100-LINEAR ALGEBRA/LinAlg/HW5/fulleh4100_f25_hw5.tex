\newpage\item Let $x_1,x_2,x_3\in\mathbb{R}$ and denote by $\mathbb{P}_2$ the space of all real polynomials of degree at most 2. Let $T\in\mathcal{L}(\mathbb{P}_2,\mathbb{R}^3)$ be the evaluation operator defined by $T:p(x)\rightarrow (p(x_1),p(x_2),p(x_3))$. Find the matrix representation of $T$ with respect to the standard basis, $\{e_i=x^i:i=0,1,2\}$ and show that its determinant is equal to $(x_3-x_2)(x_3-x_1)(x_2-x_1)$. 
\\
\\$V=\left[\begin{array}{ccc}1 & x_1 & x_1^2\\1 & x_2 & x_2^2\\1 & x_3 & x_3^2\end{array}\right]$
\\$\textrm{det}V=\textrm{det}\left[\begin{array}{ccc}1&x_1&x_1^2\\0&x_2-x_1&x_2^2-x_1^2\\0&x_3-x_1&x_3^2-x_1^2\end{array}\right]$
\\$\textrm{det}V=(x_2-x_1)(x_3-x_1)\textrm{det}\left[\begin{array}{ccc}1&x_1&x_1^2\\0&1&x_2+x_1\\0&1&x_3+x_1\end{array}\right]$

\\$\textrm{det}V=(x_3-x_2)(x_3-x_1)(x_2-x_1)$

\newpage\item Let $A,B,C,D$ be $n\times n$ matrices such that $A$ is invertible and consider the block matrix 
$M=\left[
\begin{array}{cc}
A & B\\
C & D\\
\end{array}\right]$.

\begin{enumerate}[a.]
\item Show that the block matrix $E=\left[
\begin{array}{cc}
I & 0\\
-CA^{-1} & I\\
\end{array}\right]$
reduces $M$ to upper-block-triangular form and deduce that det$(M)=\textrm{det}(A)\textrm{det}(D-CA^{-1}B)$
\\
\\$EM=\left[\begin{array}{cc}I & 0\\-CA^{-1} & I\end{array}\right]\left[\begin{array}{cc}A & B\\C & D\end{array}\right]=\left[\begin{array}{cc}A & B\\0 & D-CA^{-1}B\end{array}\right]$
\\$\textrm{det}(EM)=\textrm{det}(A)\textrm{det}(D-CA^{-1}B)$
\\$\textrm{det}(E)=1$
\\$\textrm{det}(M)=\textrm{det}(E)^{-1}\textrm{det}(EM)=\textrm{det}(EM)=\textrm{det}(A)\textrm{det}(D-CA^{-1}B)$
\\
\item Use the result above to compute the determinant of
$$M= \left[
\begin{array}{cccc}
1 & 3 & -2 & 1 \\
4 & 5 & 3 & -2 \\
-1 & 4 & 1 & 3 \\
2 & -1 & 4 & -5
\end{array}
\right]$$
\\
\\$A=\left[\begin{array}{cc}1 & 3\\4 & 5\end{array}\right]$, $B=\left[\begin{array}{cc}-2 & 1\\3 & -2\end{array}\right]$
\\$C=\left[\begin{array}{cc}-1 & 4\\2 & -1\end{array}\right]$, $D=\left[\begin{array}{cc}1 & 3\\4 & -5\end{array}\right]$
\\$\textrm{det}A=1*5-3*4=5-12=-7$
\\$A^{-1}=\frac{1}{\textrm{det}A}\left[\begin{array}{cc}5 & -3\\-4 & 1\end{array}\right]=\left[\begin{array}{cc}-5/7 & 3/7\\4/7 & -1/7\end{array}\right]$
\\$A^{-1}B=\left[\begin{array}{cc}-5/7 & 3/7\\4/7 & -1/7\end{array}\right]\left[\begin{array}{cc}-2 & 1\\3 & -2\end{array}\right]=\left[\begin{array}{cc}19/7 & -11/7\\-11/7 & 6/7\end{array}\right]$
\\$CA^{-1}B=\left[\begin{array}{cc}-1 & 4\\2 & -1\end{array}\right]\left[\begin{array}{cc}19/7 & -11/7\\-11/7 & 6/7\end{array}\right]=\left[\begin{array}{cc}-9 & 5\\7 & -4\end{array}\right]$
\\$D-CA^{-1}B=\left[\begin{array}{cc}1 & 3\\4 & -5\end{array}\right]-\left[\begin{array}{cc}-9 & 5\\7 & -4\end{array}\right]=\left[\begin{array}{cc}10 & -2\\-3 & -1\end{array}\right]$
\\$\textrm{det}D-CA^{-1}B=10*(-1) - (-2)*(-3)=-10-6=-16$
\\$\textrm{det}M=\textrm{det}A\textrm{det}D-CA^{-1}B=(-7)*(-16)=112$

\end{enumerate}
