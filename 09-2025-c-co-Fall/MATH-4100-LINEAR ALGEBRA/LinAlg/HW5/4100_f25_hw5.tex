\begin{enumerate}

\item  Let 
$$A= \left[
\begin{array}{ccc}
2 & -1 & 0 \\
4 & -1 & 3 \\
-6 & 1 & -10
\end{array} \right] $$

\bigskip

find the $LU$ factorization for $A$ and use it to solve the system
$$\begin{array}{lcr}
2x_1-x_2 & = & 6 \\
4x_1-x_2+3x_3 & = & 1 \\
-6x_1+x_2-10x_3 & = & 16
\end{array}$$

\item Find the $QR$ factorization of the matrix
$$A= \left[
\begin{array}{cc}
-1 & 3 \\
0 & 1 \\
1 & 0 
\end{array} \right] $$ 

and use it to find the least squares solution to $A{\bf x}=\left[\begin{array}{c}
2\\
1\\
2\\
\end{array}\right]$

\item Let $A\in\mathbb{R}^{2\times 2}$ such that $A^2-A=6I$ and $|A-3I|=|A+2I|=0$. Compute the determminant of $A$ without using eigenvalues.

\item Let $x_1,x_2,x_3\in\mathbb{R}$ and denote by $\mathbb{P}_2$ the space of all real polynomials of degree at most 2. Let $T\in\mathcal{L}(\mathbb{P}_2,\mathbb{R}^3)$ be the evaluation operator defined by $T:p(x)\rightarrow (p(x_1),p(x_2),p(x_3))$. Find the matrix representation of $T$ with respect to the standard basis, $\{e_i=x^i:i=0,1,2\}$ and show that its determinant is equal to $(x_3-x_2)(x_3-x_1)(x_2-x_1)$. 

\item Let $A,B,C,D$ be $n\times n$ matrices such that $A$ is invertible and consider the block matrix 
$M=\left[
\begin{array}{cc}
A & B\\
C & D\\
\end{array}\right]$.

\begin{enumerate}[a.]
\item Show that the block matrix $E=\left[
\begin{array}{cc}
I & 0\\
-CA^{-1} & I\\
\end{array}\right]$
reduces $M$ to upper-block-triangular form and deduce that det$(M)=\textrm{det}(A)\textrm{det}(D-CA^{-1}B)$

\item Use the result above to compute the determinant of
$$M= \left[
\begin{array}{cccc}
1 & 3 & -2 & 1 \\
4 & 5 & 3 & -2 \\
-1 & 4 & 1 & 3 \\
2 & -1 & 4 & -5
\end{array}
\right]$$

\end{enumerate}

\item Use Cramer's rule to solve the system
$$\begin{array}{lcr}
2x_1+3x_2+x_3 & = & 8 \\
4x_1+7x_2+5x_3 & = & 20 \\
-2x_2+2x_3 & = & 0
\end{array}$$


\end{enumerate}
